# 6. dynamic_tweights (Dynamic Networks & Transmission Weights (dynamic_tweights))

Same model as the dynamic_tweights example of `Documentation/31. Network-Based Diffusion Analysis.qmd`, on `data/nbda_data.json`.

```python
from BayesForge import bf
import json
import jax.numpy as jnp

m = bf(platform='cpu')

# STbayes example data (50 individuals, 1 trial, 1 network), prepared for BF in nbda_data.json:
# A (network: trials x networks x time x P x P), Z (informed individuals), D (durations),
# obs_end_time / is_event / valid_ind (time of acquisition or censoring of each individual)
with open('nbda_data.json') as f:
    data_list = {k: jnp.array(v) if isinstance(v, list) else v for k, v in json.load(f).items()}
data_list['Zn'] = data_list['Z']  # informed individuals seen by the naive ones

import jax

m.data_on_model = {'data': data_list}

def model_dynamic_tweights(data):
    log_lambda_0_mean = m.dist.normal(-4, 2, name="log_lambda_0_mean")
    log_s_prime_mean = m.dist.normal(-4, 2, name="log_s_prime_mean")
    k_raw = m.dist.normal(0, 3, name="k_raw")

    lambda_0 = jnp.exp(log_lambda_0_mean)
    s_prime = jnp.exp(log_s_prime_mean)
    k_shape = 2.0 * jax.nn.sigmoid(k_raw) - 1.0   # shape of the transmission curve, in (-1, 1)

    def dini_func(x, k):
        """Dini curve: maps the proportion of informed contacts to a transmission weight."""
        x_transformed = 2.0 * x - 1.0
        return ((x_transformed - k * x_transformed)
                / (k - 2.0 * k * jnp.abs(x_transformed) + 1.0) + 1.0) / 2.0

    Z_exp = jnp.expand_dims(jnp.expand_dims(data['Z'], axis=-1), axis=0)
    Zn_exp = jnp.expand_dims(jnp.expand_dims(data['Zn'], axis=-1), axis=0)
    numer = jnp.squeeze(jnp.matmul(data['A'], Z_exp), axis=-1)
    denom = numer + jnp.squeeze(jnp.matmul(data['A'], 1.0 - Zn_exp), axis=-1)
    prop = jnp.where(denom > 0, numer / denom, 0.0)
    soc_term = s_prime * jnp.sum(dini_func(prop, k_shape), axis=0)

    D_exp = jnp.expand_dims(data['D'], axis=-1)
    lambda_all = (lambda_0 + soc_term) * D_exp

    time_grid_exp = jnp.arange(data['T_max'])[None, :, None]
    obs_end_time_exp = jnp.expand_dims(data['obs_end_time'], axis=1)
    sum_mask = time_grid_exp < obs_end_time_exp
    log_mask = (time_grid_exp == (obs_end_time_exp - 1)) & jnp.expand_dims(data['is_event'], axis=1)
    neg_lambda_sum = jnp.sum(-lambda_all * sum_mask, axis=1)
    log_lambda_term = jnp.sum(jnp.log(lambda_all) * log_mask, axis=1)
    total_log_lik = jnp.sum((neg_lambda_sum + log_lambda_term) * data['valid_ind'])
    m.dist.unit(log_factor=total_log_lik, name="dynamic_tweights_lik")

m.fit(model_dynamic_tweights, num_samples=1000, num_warmup=1000, num_chains=4, seed=0, progress_bar=False)
```
