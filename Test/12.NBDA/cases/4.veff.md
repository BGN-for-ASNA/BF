# 4. veff (Varying Effects)

Same model as the veff example of `Documentation/31. Network-Based Diffusion Analysis.qmd`, on `data/nbda_data.json`. Only the population means are
compared (`params:`): the docs model is centered (`v ~ normal(0, sigma)`), so `sigma_lambda0` and `sigma_sprime`
sit in a funnel and mix poorly (ESS 30-95, r_hat 1.07-1.11 with 4 chains; 2000/2000 draws do not help, and Python
against itself moves `sigma_sprime` by 0.23 sd). Julia's libm then shifts them by Monte Carlo error, see
"Python, R and Julia: what is identical and what is not" in `Test/CROSS_LANGUAGE.md`.

```python
from BayesForge import bf
import json
import jax.numpy as jnp

m = bf(platform='cpu')

# STbayes example data (50 individuals, 1 trial, 1 network), prepared for BF in nbda_data.json:
# A (network: trials x networks x time x P x P), Z (informed individuals), D (durations),
# obs_end_time / is_event / valid_ind (time of acquisition or censoring of each individual)
with open('nbda_data.json') as f:
    data_list = {k: jnp.array(v) if isinstance(v, list) else v for k, v in json.load(f).items()}
data_list['Zn'] = data_list['Z']  # informed individuals seen by the naive ones

m.data_on_model = {'data': data_list}

def model_veff(data):
    log_lambda_0_mean = m.dist.normal(-4, 2, name="log_lambda_0_mean")
    log_s_prime_mean = m.dist.normal(-4, 2, name="log_s_prime_mean")

    # Individual offsets (varying effects) on both rates
    P = data['Z'].shape[2]
    sigma_sprime = m.dist.half_normal(1.0, name="sigma_sprime")
    sigma_lambda0 = m.dist.half_normal(1.0, name="sigma_lambda0")
    v_sprime = m.dist.normal(0.0, sigma_sprime, shape=(P,), name="v_sprime")
    v_lambda0 = m.dist.normal(0.0, sigma_lambda0, shape=(P,), name="v_lambda0")
    s_prime = jnp.exp(log_s_prime_mean + v_sprime)      # (P,) individual social rates
    lambda_0 = jnp.exp(log_lambda_0_mean + v_lambda0)   # (P,) individual intrinsic rates

    A_sum = jnp.sum(data['A'], axis=0)
    Z_exp = jnp.expand_dims(data['Z'], axis=-1)
    A_Z = jnp.squeeze(jnp.matmul(A_sum, Z_exp), axis=-1)
    soc_term = s_prime * A_Z

    D_exp = jnp.expand_dims(data['D'], axis=-1)
    lambda_all = (lambda_0 + soc_term) * D_exp

    time_grid_exp = jnp.arange(data['T_max'])[None, :, None]
    obs_end_time_exp = jnp.expand_dims(data['obs_end_time'], axis=1)
    sum_mask = time_grid_exp < obs_end_time_exp
    log_mask = (time_grid_exp == (obs_end_time_exp - 1)) & jnp.expand_dims(data['is_event'], axis=1)
    neg_lambda_sum = jnp.sum(-lambda_all * sum_mask, axis=1)
    log_lambda_term = jnp.sum(jnp.log(lambda_all) * log_mask, axis=1)
    total_log_lik = jnp.sum((neg_lambda_sum + log_lambda_term) * data['valid_ind'])
    m.dist.unit(log_factor=total_log_lik, name="veff_lik")

m.fit(model_veff, num_samples=1000, num_warmup=1000, num_chains=4, seed=0, progress_bar=False)
```
