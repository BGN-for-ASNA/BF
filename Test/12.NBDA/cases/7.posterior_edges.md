# 7. posterior_edges (Network Uncertainty (posterior_edges))

Same model as the posterior_edges example of `Documentation/31. Network-Based Diffusion Analysis.qmd`, on `data/nbda_data.json`.

```python
from BayesForge import bf
import json
import jax.numpy as jnp

m = bf(platform='cpu')

# STbayes example data (50 individuals, 1 trial, 1 network), prepared for BF in nbda_data.json:
# A (network: trials x networks x time x P x P), Z (informed individuals), D (durations),
# obs_end_time / is_event / valid_ind (time of acquisition or censoring of each individual)
with open('nbda_data.json') as f:
    data_list = {k: jnp.array(v) if isinstance(v, list) else v for k, v in json.load(f).items()}
data_list['Zn'] = data_list['Z']  # informed individuals seen by the naive ones

import jax

# Edge weights known only up to a posterior (e.g. from an edge model, see Network Edges Inference):
# here a standard normal on the logit scale for each of the P(P-1)/2 dyads
P = data_list['P']
focal_ID, other_ID = jnp.triu_indices(P, k=1)
N_dyad = focal_ID.shape[0]
data_list['N_dyad'] = N_dyad
data_list['logit_edge_mu'] = jnp.zeros((1, N_dyad))
data_list['logit_edge_cov'] = jnp.eye(N_dyad)[None]
data_list['focal_ID'] = focal_ID + 1   # 1-based ids, as STbayes
data_list['other_ID'] = other_ID + 1
m.data_on_model = {'data': data_list}

def model_posterior_edges(data):
    log_lambda_0_mean = m.dist.normal(-4, 2, name="log_lambda_0_mean")
    log_s_prime_mean = m.dist.normal(-4, 2, name="log_s_prime_mean")
    lambda_0 = jnp.exp(log_lambda_0_mean)
    s_prime = jnp.exp(log_s_prime_mean)

    # Sample the edge weights from their posterior and build the network
    edge_logit = m.dist.multivariate_normal(loc=data['logit_edge_mu'],
                                            covariance_matrix=data['logit_edge_cov'],
                                            name="edge_logit")
    w = jax.nn.sigmoid(edge_logit)
    network_idx = jnp.arange(data['N_networks'])[:, None]
    focal_idx = data['focal_ID'] - 1
    other_idx = data['other_ID'] - 1
    A = jnp.zeros((data['N_networks'], data['P'], data['P']))
    A = A.at[network_idx, focal_idx, other_idx].set(w)
    A = A.at[network_idx, other_idx, focal_idx].set(w)
    A_sum = jnp.sum(A, axis=0)[None, None]   # same network at every trial and time step

    Z_exp = jnp.expand_dims(data['Z'], axis=-1)
    A_Z = jnp.squeeze(jnp.matmul(A_sum, Z_exp), axis=-1)
    soc_term = s_prime * A_Z

    D_exp = jnp.expand_dims(data['D'], axis=-1)
    lambda_all = (lambda_0 + soc_term) * D_exp

    time_grid_exp = jnp.arange(data['T_max'])[None, :, None]
    obs_end_time_exp = jnp.expand_dims(data['obs_end_time'], axis=1)
    sum_mask = time_grid_exp < obs_end_time_exp
    log_mask = (time_grid_exp == (obs_end_time_exp - 1)) & jnp.expand_dims(data['is_event'], axis=1)
    neg_lambda_sum = jnp.sum(-lambda_all * sum_mask, axis=1)
    log_lambda_term = jnp.sum(jnp.log(lambda_all) * log_mask, axis=1)
    total_log_lik = jnp.sum((neg_lambda_sum + log_lambda_term) * data['valid_ind'])
    m.dist.unit(log_factor=total_log_lik, name="posterior_edges_lik")

m.fit(model_posterior_edges, num_samples=1000, num_warmup=1000, num_chains=4, seed=0, progress_bar=False)
```
