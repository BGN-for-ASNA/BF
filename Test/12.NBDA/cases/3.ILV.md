# 3. ILV (Individual-Level Variables)

Same model as the ILV example of `Documentation/31. Network-Based Diffusion Analysis.qmd`, on `data/nbda_data.json`.

```python
from BayesForge import bf
import json
import jax.numpy as jnp

m = bf(platform='cpu')

# STbayes example data (50 individuals, 1 trial, 1 network), prepared for BF in nbda_data.json:
# A (network: trials x networks x time x P x P), Z (informed individuals), D (durations),
# obs_end_time / is_event / valid_ind (time of acquisition or censoring of each individual)
with open('nbda_data.json') as f:
    data_list = {k: jnp.array(v) if isinstance(v, list) else v for k, v in json.load(f).items()}
data_list['Zn'] = data_list['Z']  # informed individuals seen by the naive ones

# One individual-level variable (ILV) used for asocial, social and multiplicative effects
data_list['ILV_bool_ILV'] = data_list['ILV'][:, None]                                      # (P, 1)
data_list['ILV_cont_ILV'] = data_list['ILV']                                               # (P,)
data_list['ILV_cat_ILV'] = jnp.zeros((data_list['P'], 3)).at[:, 0].set(data_list['ILV'])  # (P, 3)
m.data_on_model = {'data': data_list}

def model_ILV(data):
    log_lambda_0_mean = m.dist.normal(-4, 2, name="log_lambda_0_mean")
    log_s_prime_mean = m.dist.normal(-4, 2, name="log_s_prime_mean")
    beta_ILVi_bool_ILV = m.dist.normal(0, 1, shape=(1,), name="beta_ILVi_bool_ILV")
    beta_ILVs_cont_ILV = m.dist.normal(0, 1, name="beta_ILVs_cont_ILV")
    beta_ILVm_cat_ILV = m.dist.normal(0, 1, shape=(3,), name="beta_ILVm_cat_ILV")

    lambda_0 = jnp.exp(log_lambda_0_mean)
    s_prime = jnp.exp(log_s_prime_mean)

    # ILV effects on asocial learning (i), social learning (s), and both (m)
    ind_term = jnp.exp(jnp.matmul(data['ILV_bool_ILV'], beta_ILVi_bool_ILV))
    cont_ILV_s = data['ILV_cont_ILV'] * beta_ILVs_cont_ILV
    cat_ILV_m = jnp.matmul(data['ILV_cat_ILV'], beta_ILVm_cat_ILV)

    A_sum = jnp.sum(data['A'], axis=0)
    Z_exp = jnp.expand_dims(data['Z'], axis=-1)
    A_Z = jnp.squeeze(jnp.matmul(A_sum, Z_exp), axis=-1)
    soc_term = s_prime * A_Z * jnp.exp(cont_ILV_s)

    D_exp = jnp.expand_dims(data['D'], axis=-1)
    lambda_all = jnp.exp(cat_ILV_m) * (lambda_0 * ind_term + soc_term) * D_exp

    time_grid_exp = jnp.arange(data['T_max'])[None, :, None]
    obs_end_time_exp = jnp.expand_dims(data['obs_end_time'], axis=1)
    sum_mask = time_grid_exp < obs_end_time_exp
    log_mask = (time_grid_exp == (obs_end_time_exp - 1)) & jnp.expand_dims(data['is_event'], axis=1)
    neg_lambda_sum = jnp.sum(-lambda_all * sum_mask, axis=1)
    log_lambda_term = jnp.sum(jnp.log(lambda_all) * log_mask, axis=1)
    total_log_lik = jnp.sum((neg_lambda_sum + log_lambda_term) * data['valid_ind'])
    m.dist.unit(log_factor=total_log_lik, name="ILV_lik")

m.fit(model_ILV, num_samples=1000, num_warmup=1000, num_chains=4, seed=0, progress_bar=False)
```
