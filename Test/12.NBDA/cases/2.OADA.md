# 2. OADA (Order of Acquisition Data Analysis)

Same model as the OADA example of `Documentation/31. Network-Based Diffusion Analysis.qmd`, on `data/nbda_data.json`.

```python
from BayesForge import bf
import json
import jax.numpy as jnp

m = bf(platform='cpu')

# STbayes example data (50 individuals, 1 trial, 1 network), prepared for BF in nbda_data.json:
# A (network: trials x networks x time x P x P), Z (informed individuals), D (durations),
# obs_end_time / is_event / valid_ind (time of acquisition or censoring of each individual)
with open('nbda_data.json') as f:
    data_list = {k: jnp.array(v) if isinstance(v, list) else v for k, v in json.load(f).items()}
data_list['Zn'] = data_list['Z']  # informed individuals seen by the naive ones

m.data_on_model = {'data': data_list}

def model_OADA(data):
    log_s_prime_mean = m.dist.normal(-4, 2, name="log_s_prime_mean")
    s_prime = jnp.exp(log_s_prime_mean)

    A_sum = jnp.sum(data['A'], axis=0)
    Z_exp = jnp.expand_dims(data['Z'], axis=-1)
    A_Z = jnp.squeeze(jnp.matmul(A_sum, Z_exp), axis=-1)
    lambda_all = 1.0 + s_prime * A_Z   # relative rate: asocial rate fixed to 1

    # Probability that the individual who learnt at each step was the one to learn
    i_lambda = jnp.sum(lambda_all * data['is_event_3d'], axis=-1)              # (K, T_max)
    Q_mask = jnp.arange(data['P']) < data['Q']
    j_rates_sum = jnp.sum(lambda_all * (1.0 - data['Z']) * Q_mask, axis=-1)   # naive individuals
    event_at_time = data['event_at_time']
    log_lik = (jnp.log(jnp.where(event_at_time, i_lambda, 1.0))
               - jnp.log(jnp.where(event_at_time, j_rates_sum, 1.0)))
    total_log_lik = jnp.sum(jnp.where(event_at_time, log_lik, 0.0))
    m.dist.unit(log_factor=total_log_lik, name="OADA_lik")

m.fit(model_OADA, num_samples=1000, num_warmup=1000, num_chains=4, seed=0, progress_bar=False)
```
