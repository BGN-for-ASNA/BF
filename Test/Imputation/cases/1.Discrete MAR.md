# 1. Discrete imputation, missing at random (binary covariate marginalised out)

Same model as `model_no_plate` in `../0.Code.py` (NumPyro discrete imputation tutorial), on data simulated once by `data/make_data.py`.

```python
from BayesForge import bf
import jax.numpy as jnp

m = bf(platform='cpu')
m.data('imputation_mar.csv', sep=',')
m.data_on_model = {
    'A': jnp.array(m.df.A.values, dtype=jnp.int32),
    'B': jnp.array(m.df.B.values, dtype=jnp.int32),
    'Y': jnp.array(m.df.Y.values),
}

def model(A, B, Y):
    A_isobs = A >= 0
    mu_A = m.dist.normal(0, 2.5, name='mu_A')
    b_B_A = m.dist.normal(0, 2.5, name='b_B_A')
    b_A = m.dist.normal(0, 2.5, name='b_A')
    b_B = m.dist.normal(0, 2.5, name='b_B')
    s_Y = m.dist.exponential(1.0, name='s_Y')
    eta_A = mu_A + b_B_A * B
    dist_A = m.dist.bernoulli(logits=eta_A, create_obj=True)
    A_valid = jnp.maximum(A, 0)
    ll_obs = dist_A.log_prob(A_valid) + m.dist.normal(b_A * A_valid + b_B * B, s_Y, create_obj=True).log_prob(Y)
    ll_0 = dist_A.log_prob(0) + m.dist.normal(b_B * B, s_Y, create_obj=True).log_prob(Y)
    ll_1 = dist_A.log_prob(1) + m.dist.normal(b_A + b_B * B, s_Y, create_obj=True).log_prob(Y)
    ll_mis = jnp.logaddexp(ll_0, ll_1)
    m.dist.factor("log_lik", jnp.sum(jnp.where(A_isobs, ll_obs, ll_mis)))

m.fit(model, num_samples=1000, num_warmup=1000, num_chains=4, seed=0, progress_bar=False)
```
