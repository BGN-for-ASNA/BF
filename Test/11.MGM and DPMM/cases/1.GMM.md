# 1. Gaussian mixture model (K = 8, means anchored on frozen k-means centroids)

Same model as `Documentation/24. GMM.qmd` and `../Similarity_Test.py`, on blobs and k-means centroids frozen by `data/make_data.py`. The tight prior around the centroids (covariance 0.1) fixes the component labels, so the components can be compared one by one.

```python
from BayesForge import bf
import pandas as pd
import jax.numpy as jnp

m = bf(platform='cpu')

blobs = pd.read_csv("blobs.csv")
centroids = pd.read_csv("gmm_centroids.csv")
data = jnp.array(blobs.values)
initial_means = jnp.array(centroids.values)
K = 8

m.data_on_model = dict(data=data, K=8, initial_means=initial_means)

def gmm(data, K, initial_means): # Here K is the *exact* number of clusters
    D = data.shape[1]  # Number of features
    alpha_prior = 0.5 * jnp.ones(K)
    w = m.dist.dirichlet(concentration=alpha_prior, name='weights') 

    with m.dist.plate("components", K): # Use fixed K
        mu = m.dist.multivariate_normal(loc=initial_means, covariance_matrix=0.1*jnp.eye(D), name='mu')        
        sigma = m.dist.half_cauchy(1, shape=(D,), event=1, name='sigma')
        Lcorr = m.dist.lkj_cholesky(dimension=D, concentration=1.0, name='Lcorr')

        scale_tril = sigma[..., None] * Lcorr

    m.dist.mixture_same_family(
        mixing_distribution=m.dist.categorical(probs=w, create_obj=True),
        component_distribution=m.dist.multivariate_normal(loc=mu, scale_tril=scale_tril, create_obj=True),
        name="obs",
        obs=data
    )

m.fit(gmm, num_samples=1000, num_warmup=1000, num_chains=4, seed=0, progress_bar=False)
```

library(BayesForge)
m <- importBF(platform = "cpu")
pd <- reticulate::import("pandas", convert = FALSE)
jnp <- reticulate::import("jax.numpy")

blobs <- pd$read_csv("blobs.csv")
centroids <- pd$read_csv("gmm_centroids.csv")
data <- jnp$array(blobs$values)
initial_means <- jnp$array(centroids$values)
K <- 8

m$data_on_model <- list(data = data, K = 8L, initial_means = initial_means)

model_gmm <- function(data, K, initial_means) { # Here K is the *exact* number of clusters
  D <- data$shape[[2]]  # Number of features
  alpha_prior <- 0.5 * jnp$ones(K)
  w <- m$dist$dirichlet(concentration = alpha_prior, name = "weights")

  with(m$dist$plate("components", K), {
    mu <- m$dist$multivariate_normal(loc = initial_means, covariance_matrix = 0.1 * jnp$eye(D), name = "mu")
    sigma <- m$dist$half_cauchy(1, shape = reticulate::tuple(D), event = 1L, name = "sigma")
    Lcorr <- m$dist$lkj_cholesky(dimension = D, concentration = 1.0, name = "Lcorr")
    scale_tril <- jnp$expand_dims(sigma, -1L) * Lcorr
  })

  m$dist$mixture_same_family(
    mixing_distribution = m$dist$categorical(probs = w, create_obj = TRUE),
    component_distribution = m$dist$multivariate_normal(loc = mu, scale_tril = scale_tril, create_obj = TRUE),
    name = "obs",
    obs = data
  )
}

m$fit(model_gmm, num_samples = 1000L, num_warmup = 1000L, num_chains = 4L, seed = 0L, progress_bar = FALSE)
