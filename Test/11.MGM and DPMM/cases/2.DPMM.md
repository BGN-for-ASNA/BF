# 2. Dirichlet process mixture (stick-breaking, K = 10, blobs data)

Same model as `Documentation/25. DPMM.qmd` and `../Similarity_Test.py`. Only the concentration `alpha` is compared (`params:`): all components share one prior, so their labels can switch between chains and languages. `tol` is 0.5
because `alpha` mixes slowly: with 4 chains its per-chain means range 0.45-0.65 and some `beta`, `w`, `mu`
entries have r_hat up to 1.55; Python against itself with seed 1 instead of 0 moves `alpha` by 0.30 sd,
and Julia (openlibm, see "Python, R and Julia: what is identical and what is not" in `Test/CROSS_LANGUAGE.md`)
moves it by 0.40 sd.

```python
from BayesForge import bf
import pandas as pd
import jax.numpy as jnp
import numpyro

m = bf(platform='cpu')

blobs = pd.read_csv("blobs.csv")
data = jnp.array(blobs.values)
data_mean = jnp.mean(data, axis=0)
data_std = jnp.std(data, axis=0) * 2
K = 10

m.data_on_model = dict(data=data, K=10, data_mean=data_mean, data_std=data_std)

def dpmm(data, K, data_mean, data_std):
    N, D = data.shape  # Number of features


    # 1) stick-breaking weights
    alpha = m.dist.gamma(1.0, 10.0,name='alpha')

    with m.dist.plate("beta_plate", K - 1):
        beta = m.dist.beta(1, alpha, name = "beta")

    w = numpyro.deterministic("w",m.models.dpmm.mix_weights(beta))

    # 2) component parameters
    with m.dist.plate("components", K):
        mu = m.dist.multivariate_normal(loc=data_mean, covariance_matrix=data_std*jnp.eye(D),name='mu')# shape (T, D)        
        sigma = m.dist.log_normal(0.0, 1.0,shape=(D,),event=1,name='sigma')# shape (T, D)
        Lcorr = m.dist.lkj_cholesky(dimension=D, concentration=1.0,name='Lcorr')# shape (T, D, D)

        scale_tril = sigma[..., None] * Lcorr  # shape (T, D, D)

    # 3) Latent cluster assignments for each data point
    m.dist.mixture_same_family(
        mixing_distribution=m.dist.categorical(probs=w, create_obj=True),
        component_distribution=m.dist.multivariate_normal(
            loc=mu, 
            scale_tril=scale_tril, 
            create_obj=True
        ),
        name="obs",
        obs=data
    )

m.fit(dpmm, num_samples=1000, num_warmup=1000, num_chains=4, seed=0, progress_bar=False)
```

library(BayesForge)
m <- importBF(platform = "cpu")
pd <- reticulate::import("pandas", convert = FALSE)
jnp <- reticulate::import("jax.numpy")
numpyro <- reticulate::import("numpyro")

blobs <- pd$read_csv("blobs.csv")
data <- jnp$array(blobs$values)
data_mean <- jnp$mean(data, axis = 0L)
data_std <- jnp$std(data, axis = 0L) * 2
K <- 10

m$data_on_model <- list(data = data, K = 10L, data_mean = data_mean, data_std = data_std)

model_dpmm <- function(data, K, data_mean, data_std) {
  N <- data$shape[[1]]
  D <- data$shape[[2]]  # Number of features

  # 1) stick-breaking weights
  alpha <- m$dist$gamma(1.0, 10.0, name = "alpha")

  with(m$dist$plate("beta_plate", K - 1L), {
    beta <- m$dist$beta(1, alpha, name = "beta")
  })

  w <- numpyro$deterministic("w", m$models$dpmm$mix_weights(beta))

  # 2) component parameters
  with(m$dist$plate("components", K), {
    mu <- m$dist$multivariate_normal(loc = data_mean, covariance_matrix = data_std * jnp$eye(D), name = "mu") # shape (T, D)
    sigma <- m$dist$log_normal(0.0, 1.0, shape = reticulate::tuple(D), event = 1L, name = "sigma") # shape (T, D)
    Lcorr <- m$dist$lkj_cholesky(dimension = D, concentration = 1.0, name = "Lcorr") # shape (T, D, D)
    scale_tril <- jnp$expand_dims(sigma, -1L) * Lcorr # shape (T, D, D)
  })

  # 3) Latent cluster assignments for each data point
  m$dist$mixture_same_family(
    mixing_distribution = m$dist$categorical(probs = w, create_obj = TRUE),
    component_distribution = m$dist$multivariate_normal(
      loc = mu,
      scale_tril = scale_tril,
      create_obj = TRUE
    ),
    name = "obs",
    obs = data
  )
}

m$fit(model_dpmm, num_samples = 1000L, num_warmup = 1000L, num_chains = 4L, seed = 0L, progress_bar = FALSE)
