# 1. Repertoire size (zero-inflated Poisson token counts, 50 individuals x 50 tokens)

Same model as `../scripts/BF_only.py` (McElreath et al. repertoire model), data `stan_data.json` (simulated by `../scripts/generate_data.r`).

```python
from BayesForge import bf
import json
import jax.numpy as jnp

m = bf(platform='cpu')

# Repertoire data (504 observations, M = 50 tokens), prepared for Stan in stan_data.json:
# d (per-observation exposure), Y (token counts, J x M = 504 x 50)
with open('stan_data.json') as f:
    data = json.load(f)
M = 50
d = jnp.array(data['d'])
Y = jnp.array(data['Y'], dtype=jnp.int32)

m.data_on_model = dict(M=50, d=d, Y=Y)

def model(M, d, Y):
    L = m.dist.exponential(1.0, shape=(M,), name='L')          # rate of each token
    p = m.dist.beta(2.0, 2.0, shape=(M,), name='p')            # population probability of each token
    rate = jnp.expand_dims(L, 0) * jnp.expand_dims(d, 1)       # J x M
    gate = 1.0 - jnp.expand_dims(p, 0)                         # probability of a structural zero
    m.dist.zero_inflated_poisson(gate=gate, rate=rate, obs=Y, name='Y')

m.fit(model, num_samples=1000, num_warmup=1000, num_chains=4, seed=0, progress_bar=False)
```

}
