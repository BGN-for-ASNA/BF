# 1. Nested varying effects (regions > groups, correlated intercepts and slopes, built-in non-centered)

Same model as the built-in example of `Documentation/17. Nested varying effects.qmd` (dataset `m.load.sim_nested_effects`; see also `../builtin_function.py`).

```python
from BayesForge import bf
import jax.numpy as jnp

# Setup device -----------------------------------------------
m = bf(platform='cpu')

# Load data — column args are mapped automatically from the DataFrame.
# N_groups and N_regions are dataset-specific; provide them as defaults.
data_path = m.load.sim_nested_effects(only_path=True)
m.data(data_path)

# group_ids: one obs-level index array per level, top to bottom.
# Parent structure is derived automatically from these arrays.
def model_nested_builtin(y, x, group_id, region_id, N_groups=20, N_regions=5):
    sigma = m.dist.exponential(1, name='sigma')

    a_g_est, b_g_est = m.effects.nested_varying_effects(
        N_vars=2,
        names=["region", "group"],
        N_groups=[N_regions, N_groups],
        group_ids=[region_id, group_id],
        centered=False,
    )

    mu_est = a_g_est + b_g_est * x
    m.dist.normal(mu_est, sigma, obs=y, name='y')

m.fit(model_nested_builtin, num_samples=1000, num_warmup=1000, num_chains=4, seed=0, progress_bar=False)
```

```r
library(BayesForge)

# Setup platform ------------------------------------------------
m = importBF(platform='cpu')
jnp = reticulate::import("jax.numpy")

# Load data — column args mapped automatically from the DataFrame.
# N_groups and N_regions are dataset-specific; provide them as defaults.
data_path = m$load$sim_nested_effects(only_path=TRUE)
m$data(data_path)

# The R wrapper exposes no Python-visible defaults, so pass the model
# data (DataFrame columns + scalar sizes) explicitly.
m$data_on_model = list(
  y         = jnp$array(m$df$y),
  x         = jnp$array(m$df$x),
  group_id  = jnp$array(as.integer(m$df$group_id)),
  region_id = jnp$array(as.integer(m$df$region_id)),
  N_groups  = 20L,
  N_regions = 5L
)

# Define model ------------------------------------------------
model <- function(y, x, group_id, region_id, N_groups=20L, N_regions=5L) {
  sigma = m$dist$exponential(1, name='sigma')

  veff = m$effects$nested_varying_effects(
    N_vars    = 2L,
    names     = list('region', 'group'),
    N_groups  = list(as.integer(N_regions), as.integer(N_groups)),
    group_ids = list(region_id, group_id),
    centered  = FALSE,
  )
  a_g_est = veff[[1]]
  b_g_est = veff[[2]]

  mu = a_g_est + b_g_est * x
  m$dist$normal(mu, sigma, obs=y, name='y')
}

# Run MCMC ------------------------------------------------
m$fit(model, num_samples = 1000L, num_warmup = 1000L, num_chains = 4L, seed = 0L, progress_bar = FALSE)
```
