# 8b. Multinomial model (counts across categories with covariate X)

Same model as `../Numpyro backend/8b.Multinomial fixed income.py` and `Documentation/11. Multinomial model.qmd`, in the three languages.

```python
from BayesForge import bf
import jax.numpy as jnp
import jax
import numpy as np

m = bf('cpu')
np.random.seed(42)
N = 200
n_trials = 20
X = np.random.normal(0, 1, size=N)
phi = np.column_stack([0.5 + 1.0 * X, -0.5 + 0.5 * X, np.zeros(N)])
theta = np.asarray(jax.nn.softmax(jnp.array(phi), axis=1))
Y = np.array([np.random.multinomial(n_trials, theta[i]) for i in range(N)])
m.data_on_model = dict(X=jnp.array(X), Y=jnp.array(Y))

def model(X, Y):
    alpha = m.dist.normal(0, 1, shape=(2,), name='alpha')
    beta = m.dist.normal(0, 1, shape=(2,), name='beta')
    s1 = alpha[0] + beta[0] * X
    s2 = alpha[1] + beta[1] * X
    s3 = jnp.zeros_like(s1)
    phi = jnp.stack([s1, s2, s3], axis=1)
    theta = jax.nn.softmax(phi, axis=1)
    m.dist.multinomial(total_count=20, probs=theta, obs=Y)

m.fit(model, progress_bar=False)
```

```r
library(BayesForge)
jax <- reticulate::import("jax")
jnp <- reticulate::import("jax.numpy")
m <- importBF(platform = "cpu")

set.seed(42)
N <- 200L
n_trials <- 20L
X <- rnorm(N, 0, 1)
phi <- cbind(0.5 + 1.0 * X, -0.5 + 0.5 * X, rep(0.0, N))
exp_phi <- exp(phi)
theta <- exp_phi / rowSums(exp_phi)
Y <- matrix(0L, nrow = N, ncol = 3L)
for (i in 1:N) {
  Y[i, ] <- as.integer(rmultinom(1, size = n_trials, prob = theta[i, ]))
}
m$data_on_model <- list(X = jnp$array(as.numeric(X)), Y = jnp$array(Y))

model <- function(X, Y) {
  alpha <- m$dist$normal(0, 1, name = "alpha", shape = reticulate::tuple(2L))
  beta <- m$dist$normal(0, 1, name = "beta", shape = reticulate::tuple(2L))
  s1 <- alpha[0L] + beta[0L] * X
  s2 <- alpha[1L] + beta[1L] * X
  s3 <- jnp$zeros_like(s1)
  phi <- jnp$stack(list(s1, s2, s3), axis = 1L)
  theta <- jax$nn$softmax(phi, axis = 1L)
  m$dist$multinomial(total_count = 20L, probs = theta, obs = Y, name = "Y")
}

m$fit(model, progress_bar = FALSE)
```

```julia
using PyBayesForge
using Random
using Distributions

m = importBF(platform="cpu")
Random.seed!(42)
N = 200
n_trials = 20
X = rand(Normal(0, 1), N)
Y = zeros(Int, N, 3)
for i in 1:N
    e = exp.([0.5 + 1.0 * X[i], -0.5 + 0.5 * X[i], 0.0])
    θ = e / sum(e)
    Y[i, :] = rand(Multinomial(n_trials, θ))
end
m.data_on_model = pydict("X" => jnp.array(X), "Y" => jnp.array(Y))

@BF function model(X, Y)
    alpha = m.dist.normal(0, 1, shape=(2,), name="alpha")
    beta = m.dist.normal(0, 1, shape=(2,), name="beta")

    s2 = alpha[1] + beta[1] * X

    phi = jnp.stack(pylist([s1, s2, s3]), axis=1)
    theta = jax.nn.softmax(phi, axis=1)

end

m.fit(model, progress_bar=false)
```
