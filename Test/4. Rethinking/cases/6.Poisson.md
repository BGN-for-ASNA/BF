# 6. Poisson (Kline, total_tools ~ exp(a[cid] + b[cid] * log-population))

Same model as `../Numpyro backend/6.Poisson.py`, in the three languages: Poisson rate is exponential in a contact-specific intercept and slope on the standardized log population.

```python
from BayesForge import bf
import numpy as np
import jax.numpy as jnp

m = bf(platform='cpu')
m.data(m.load.kline(only_path=True), sep=';')
log_pop = np.log(np.asarray(m.df.population, dtype=float))
m.df['population_scaled'] = (log_pop - log_pop.mean()) / log_pop.std(ddof=1)
m.df['cid'] = (m.df.contact == 'high').astype(int)
m.data_to_model(['cid', 'population_scaled', 'total_tools'])

def model(cid, population_scaled, total_tools):
    a = m.dist.normal(3, 0.5, shape=(2,), name='a')
    b = m.dist.normal(0, 0.5, shape=(2,), name='b')
    lambda_ = jnp.exp(a[cid] + b[cid] * population_scaled)
    m.dist.poisson(lambda_, obs=total_tools, name='total_tools')

m.fit(model, num_samples=1000, num_warmup=1000, seed=0, progress_bar=False)
```
