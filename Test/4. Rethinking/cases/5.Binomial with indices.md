# 5. Binomial with indices (Chimpanzees dataset, pulled_left ~ a[actor] + b[treatment])

Same model as `../Numpyro backend/5.Binomial with indices.py`, in the three languages.
The BF object must be named `m`; the harness writes `m.summary()` to `bf_summary.csv`.

```python
from BayesForge import bf
m = bf(platform='cpu')
m.data(m.load.chimpanzees(only_path=True), sep=';')
m.df['treatment'] = m.df.prosoc_left + 2 * m.df.condition
m.df['actor'] = m.df.actor.values - 1  # 0-indexed
m.data_to_model(['actor', 'treatment', 'pulled_left'])

def model_BF(actor, treatment, pulled_left):
    a = m.dist.normal(0, 1.5, shape=(7,), name='a')
    b = m.dist.normal(0, 0.5, shape=(4,), name='b')
    p = a[actor] + b[treatment]
    m.dist.binomial(total_count=1, logits=p, obs=pulled_left)

m.fit(model_BF, num_samples=1000, num_warmup=1000, seed=0, progress_bar=False)
```

```r
library(BayesForge)
m <- importBF(platform = "cpu")
jnp <- reticulate::import("jax.numpy")
m$data(m$load$chimpanzees(only_path = TRUE), sep = ";")
m$df$treatment <- m$df$prosoc_left + 2 * m$df$condition
m$df$actor <- m$df$actor - 1  # 0-indexed
m$data_to_model(list("actor", "treatment", "pulled_left"))

model_BF <- function(actor, treatment, pulled_left) {
  a <- m$dist$normal(0, 1.5, shape = reticulate::tuple(7L), name = "a")
  b <- m$dist$normal(0, 0.5, shape = reticulate::tuple(4L), name = "b")
  p <- a[jnp$asarray(actor, dtype = jnp$int32)] + b[jnp$asarray(treatment, dtype = jnp$int32)]
  m$dist$binomial(total_count = 1L, logits = p, obs = pulled_left)
}

m$fit(model_BF, num_samples = 1000L, num_warmup = 1000L, seed = 0L, progress_bar = FALSE)
```
