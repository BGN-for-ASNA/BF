# 11. Varying intercepts (Reedfrogs, bins varying intercepts)

Same model as `../Numpyro backend/11.Varying intercepts.py`, in the three languages.
The BF object must be named `m`; the harness writes `m.summary()` to `bf_summary.csv`.

```python
from BayesForge import bf
import jax.numpy as jnp

m = bf(platform='cpu')
m.data(m.load.reedfrogs(only_path=True), sep=';')
m.df['tank'] = jnp.arange(len(m.df))
m.data_on_model = {
    'tank': jnp.array(m.df.tank.values),
    'surv': jnp.array(m.df.surv.values),
    'density': jnp.array(m.df.density.values),
}

def model(tank, surv, density):
    sigma = m.dist.exponential(1.0, name='sigma')
    a_bar = m.dist.normal(0.0, 2.0, name='a_bar')
    alpha = m.dist.normal(a_bar, sigma, shape=(48,), name='alpha')
    p = alpha[tank]
    m.dist.binomial(total_count=density, logits=p, obs=surv)

m.fit(model, num_samples=1000, num_warmup=1000, seed=0, progress_bar=False)
```

```r
library(BayesForge)
m <- importBF(platform = "cpu")
jnp <- reticulate::import("jax.numpy")

m$data(m$load$reedfrogs(only_path = TRUE), sep = ";")
m$df$tank <- as.integer(0:(nrow(m$df) - 1))
m$data_to_model(list("tank", "surv", "density"))
m$data_on_model$tank <- m$data_on_model$tank$astype(jnp$int32)
m$data_on_model$surv <- m$data_on_model$surv$astype(jnp$int32)
m$data_on_model$density <- m$data_on_model$density$astype(jnp$int64)

model <- function(tank, surv, density) {
  sigma <- m$dist$exponential(1, name = "sigma")
  a_bar <- m$dist$normal(0, 2, name = "a_bar")
  alpha <- m$dist$normal(a_bar, sigma, shape = reticulate::tuple(48L), name = "alpha")
  p <- alpha$`__getitem__`(tank)
  m$dist$binomial(total_count = density, logits = p, obs = surv)
}

m$fit(model, num_samples = 1000L, num_warmup = 1000L, seed = 0L, progress_bar = FALSE)
```
