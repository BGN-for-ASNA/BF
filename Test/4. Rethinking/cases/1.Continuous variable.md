# 1. Continuous variable (Howell1 adults, height ~ weight)

Same model as `../Numpyro backend/1.Continuous variable.py`, in the three languages.
The BF object must be named `m`; the harness writes `m.summary()` to `bf_summary.csv`.

```python
from BayesForge import bf
m = bf(platform='cpu')
m.data(m.load.howell1(only_path=True), sep=';')
m.df = m.df[m.df.age > 18]
m.scale(['weight'])

def model(weight, height):
    a = m.dist.normal(178, 20, name='a')
    b = m.dist.log_normal(0, 1, name='b')
    s = m.dist.uniform(0, 50, name='s')
    m.dist.normal(a + b * weight, s, obs=height)

m.fit(model, num_samples=1000, num_warmup=1000, seed=0, progress_bar=False)
```

```r
library(BayesForge)
m <- importBF(platform = "cpu")
m$data(m$load$howell1(only_path = TRUE), sep = ";")
m$df <- m$df[m$df$age > 18, ]
m$scale(list("weight"))
m$data_to_model(list("weight", "height"))

model <- function(weight, height) {
  a <- m$dist$normal(178, 20, name = "a")
  b <- m$dist$log_normal(0, 1, name = "b")
  s <- m$dist$uniform(0, 50, name = "s")
  m$dist$normal(a + b * weight, s, obs = height)
}

m$fit(model, num_samples = 1000L, num_warmup = 1000L, seed = 0L, progress_bar = FALSE)
```
