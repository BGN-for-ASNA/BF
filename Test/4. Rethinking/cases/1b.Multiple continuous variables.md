# 1b. Multiple continuous variables (Howell1 adults, height ~ weight + age)

Same model as `../Numpyro backend/1b.Multiple continuous variables.py`, in the three languages.
The BF object must be named `m`; the harness writes `m.summary()` to `bf_summary.csv`.

```python
from BayesForge import bf
m = bf(platform='cpu')
m.data(m.load.howell1(only_path=True), sep=';')
m.df = m.df[m.df.age > 18]
m.scale(['weight', 'age'])

def model(height, weight, age):
    alpha = m.dist.normal(178, 20, name='alpha')
    beta1 = m.dist.normal(0, 0.5, name='beta1')
    beta2 = m.dist.normal(0, 0.5, name='beta2')
    sigma = m.dist.uniform(0, 50, name='sigma')
    m.dist.normal(alpha + beta1 * weight + beta2 * age, sigma, obs=height)

m.fit(model, num_samples=1000, num_warmup=1000, seed=0, progress_bar=False)
```

```r
library(BayesForge)
m <- importBF(platform = "cpu")
m$data(m$load$howell1(only_path = TRUE), sep = ";")
m$df <- m$df[m$df$age > 18, ]
m$scale(list("weight", "age"))
m$data_to_model(list("weight", "height", "age"))

model <- function(height, weight, age) {
  alpha <- m$dist$normal(178, 20, name = "alpha")
  beta1 <- m$dist$normal(0, 0.5, name = "beta1")
  beta2 <- m$dist$normal(0, 0.5, name = "beta2")
  sigma <- m$dist$uniform(0, 50, name = "sigma")
  m$dist$normal(alpha + beta1 * weight + beta2 * age, sigma, obs = height)
}

m$fit(model, num_samples = 1000L, num_warmup = 1000L, seed = 0L, progress_bar = FALSE)
```
