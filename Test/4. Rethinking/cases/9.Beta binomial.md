# 9. Beta binomial (UCBadmit, admissions ~ gender)

Same model as `../Numpyro backend/9.Beta binomial.py`, in the three languages.
The BF object must be named `m`; the harness writes `m.summary()` to `bf_summary.csv`.

```python
from BayesForge import bf
import jax.numpy as jnp

m = bf(platform='cpu')
m.data(m.load.ucbadmit(only_path=True), sep=';')

m.df['gid'] = (m.df['applicant.gender'] != 'male').astype(int)
m.data_on_model = {
    'gid': jnp.array(m.df.gid.values),
    'applications': jnp.array(m.df.applications.values),
    'admit': jnp.array(m.df.admit.values),
}

def model(gid, applications, admit):
    phi = m.dist.uniform(0.0, 20.0, name='phi')
    alpha = m.dist.normal(0.0, 1.5, shape=(2,), name='alpha')
    theta = phi + 2.0
    pbar = m.link.inv_logit(alpha[gid])
    concentration1 = pbar * theta
    concentration0 = (1.0 - pbar) * theta
    m.dist.beta_binomial(total_count=applications, concentration1=concentration1,
                         concentration0=concentration0, obs=admit)

m.fit(model, num_samples=1000, num_warmup=1000, seed=0, progress_bar=False)
```

library(BayesForge)
m <- importBF(platform = "cpu")
jnp = reticulate::import("jax.numpy")

m$data(m$load$ucbadmit(only_path = TRUE), sep = ";")
m$data_on_model$gid = jnp$array(as.integer(ifelse(m$df["applicant.gender"] == "male", 0, 1)))
m$data_on_model$applications = jnp$array(as.integer(unlist(m$df["applications"])))
m$data_on_model$admit = jnp$array(as.integer(unlist(m$df["admit"])))

model <- function(gid, applications, admit) {
  phi <- m$dist$uniform(0.0, 20.0, name = "phi")
  alpha <- m$dist$normal(0.0, 1.5, shape = reticulate::tuple(2L), name = "alpha")
  theta <- phi + 2
  pbar <- m$link$inv_logit(alpha$`__getitem__`(gid))
  concentration1 <- pbar * theta
  concentration0 <- (1 - pbar) * theta
  m$dist$beta_binomial(total_count = applications, concentration1 = concentration1,
                       concentration0 = concentration0, obs = admit)
}

m$fit(model, num_samples = 1000L, num_warmup = 1000L, seed = 0L, progress_bar = FALSE)
