# 13. Gaussian processes (Kline2, total tools ~ island distance GP)

Same model as `../Numpyro backend/13.Gaussian processes.py`, in the three languages: a non-centered
Gaussian process over the island distance matrix with a Poisson likelihood. The deterministics
`a_over_g` and `k` are reported.

```python
from BayesForge import bf
import jax.numpy as jnp
import numpyro

m = bf(platform='cpu')
m.data(m.load.kline2(only_path=True), sep=';')

islandsDistMatrix = m.load.islands_dist_matrix(frame=False).data

m.data_to_model(['total_tools', 'population'])
m.data_on_model['society'] = jnp.arange(0, 10)
m.data_on_model['Dmat'] = islandsDistMatrix

def model(Dmat, population, society, total_tools):
    a = m.dist.exponential(1, name='a')
    b = m.dist.exponential(1, name='b')
    g = m.dist.exponential(1, name='g')
    numpyro.deterministic('a_over_g', a / g)

    etasq = m.dist.exponential(2, name='etasq')
    rhosq = m.dist.exponential(0.5, name='rhosq')

    n = Dmat.shape[0]
    SIGMA = etasq * jnp.exp(-rhosq * jnp.square(Dmat))
    SIGMA = SIGMA.at[jnp.diag_indices(n)].add(0.01)
    L_chol = jnp.linalg.cholesky(SIGMA)
    z = m.dist.multivariate_normal(jnp.zeros(n), jnp.eye(n), name='z')
    k = numpyro.deterministic('k', jnp.matmul(L_chol, z))

    lambda_ = a * population**b / g * jnp.exp(k[society])
    m.dist.poisson(lambda_, obs=total_tools)

m.fit(model, num_samples=1000, num_warmup=2000, seed=0, progress_bar=False,
      target_accept_prob=0.9, dense_mass=True)
```
