# 5. Phylogenetic varying slopes (correlated phylogenetic intercepts and slopes)

Same model as `../Model_5_Slopes/fit_BF_slopes.py`, data built by `data/make_data.py`.

```python
from BayesForge import bf
import jax.numpy as jnp
import pandas as pd

m = bf(platform='cpu')
m.data('phylo_slopes.csv', sep=',')
m.data_on_model = {
    'y': jnp.array(m.df.y.values),
    'x': jnp.array(m.df.x_centered.values),
    'phylo_idx': jnp.array(m.df.phylo_idx.values, dtype=jnp.int32),
    'A_cholesky': jnp.array(pd.read_csv('phylo_L_slopes.csv', header=None).values),
}

def model(y, x, phylo_idx, A_cholesky):
    intercept = m.dist.normal(0.0, 10.0, name="Intercept")
    b_x = m.dist.normal(0.0, 10.0, name="b_x")
    sigma = m.dist.left_truncated_distribution(
        m.dist.student_t(3, 0, 2.5, create_obj=True), low=0.0, name="sigma")
    sd_intercept = m.dist.left_truncated_distribution(
        m.dist.student_t(3, 0, 10, create_obj=True), low=0.0, name="sd_intercept")
    sd_slope = m.dist.left_truncated_distribution(
        m.dist.student_t(3, 0, 10, create_obj=True), low=0.0, name="sd_slope")
    sd_1 = jnp.stack([sd_intercept, sd_slope])
    L_corr = m.dist.lkj_cholesky(2, concentration=2.0, name="L_corr")
    z_1 = m.dist.normal(jnp.zeros((100, 2)), 1.0, name="z_1")
    L_sigma = jnp.matmul(jnp.diag(sd_1), L_corr)
    U = jnp.matmul(jnp.matmul(A_cholesky, z_1), jnp.transpose(L_sigma))
    u_intercept = jnp.take(U, 0, axis=1)
    u_slope = jnp.take(U, 1, axis=1)
    mu = intercept + u_intercept[phylo_idx] + (b_x + u_slope[phylo_idx]) * x
    m.dist.normal(mu, sigma, name="Y", obs=y)

m.fit(model, num_samples=1000, num_warmup=1000, num_chains=1, seed=0, progress_bar=False)
```
