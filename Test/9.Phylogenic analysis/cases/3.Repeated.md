# 3. Phylogenetic model with repeated measurements (species and phylogenetic effects)

Same model as `../Model_3_Repeated/fit_BF_repeat.py`.

```python
from BayesForge import bf
import jax.numpy as jnp
import pandas as pd

m = bf(platform='cpu')
m.data('phylo_repeat.csv', sep=',')
m.data_on_model = {
    'phen': jnp.array(m.df.phen.values),
    'spec_mean_cf': jnp.array(m.df.spec_mean_cf_centered.values),
    'within_spec_cf': jnp.array(m.df.within_spec_cf.values),
    'phylo_idx': jnp.array(m.df.phylo_idx.values, dtype=jnp.int32),
    'species_idx': jnp.array(m.df.species_idx.values, dtype=jnp.int32),
    'A_cholesky': jnp.array(pd.read_csv('phylo_L.csv', header=None).values),
}

def model(phen, spec_mean_cf, phylo_idx, species_idx, A_cholesky, within_spec_cf):
    intercept = m.dist.normal(0, 50, name="Intercept")
    b_spec_mean_cf = m.dist.normal(0, 10, name="b_spec_mean_cf")
    b_within_spec_cf = m.dist.normal(0, 10, name="b_within_spec_cf")
    sd_phylo = m.dist.left_truncated_distribution(
        m.dist.student_t(3, 0, 20, create_obj=True), low=0.0, name="sd_phylo")
    sd_species = m.dist.left_truncated_distribution(
        m.dist.student_t(3, 0, 20, create_obj=True), low=0.0, name="sd_species")
    sigma = m.dist.left_truncated_distribution(
        m.dist.student_t(3, 0, 20, create_obj=True), low=0.0, name="sigma")
    z_phylo = m.dist.normal(jnp.zeros(200), 1.0, name="z_phylo")
    u_phylo = jnp.matmul(A_cholesky, z_phylo) * sd_phylo
    z_species = m.dist.normal(jnp.zeros(200), 1.0, name="z_species")
    u_species = z_species * sd_species
    mu = (intercept + b_spec_mean_cf * spec_mean_cf + b_within_spec_cf * within_spec_cf
          + u_phylo[phylo_idx] + u_species[species_idx])
    m.dist.normal(mu, sigma, name="obs", obs=phen)

m.fit(model, num_samples=1000, num_warmup=1000, seed=0, progress_bar=False)
```
