# 4. Phylogenetic meta-analysis (known sampling error, phylogenetic and observation-level effects)

Same model as `../Model_4_Meta/fit_BF_meta.py`, data `data_effect.txt` (Fisher's z, se = sqrt(1/(N-3))).

```python
from BayesForge import bf
import jax.numpy as jnp
import pandas as pd

m = bf(platform='cpu')
m.data('phylo_meta.csv', sep=',')
m.data_on_model = {
    'Zr': jnp.array(m.df.Zr.values),
    'se': jnp.array(m.df.se.values),
    'phylo_idx': jnp.array(m.df.phylo_idx.values, dtype=jnp.int32),
    'obs_idx': jnp.array(m.df.obs_idx.values, dtype=jnp.int32),
    'A_cholesky': jnp.array(pd.read_csv('phylo_L.csv', header=None).values),
}

def model(Zr, se, phylo_idx, obs_idx, A_cholesky):
    intercept = m.dist.normal(0.0, 10.0, name="Intercept")
    sd_obs = m.dist.left_truncated_distribution(
        m.dist.student_t(3, 0, 10, create_obj=True), low=0.0, name="sd_obs")
    sd_phylo = m.dist.left_truncated_distribution(
        m.dist.student_t(3, 0, 10, create_obj=True), low=0.0, name="sd_phylo")
    z_phylo = m.dist.normal(jnp.zeros(200), 1.0, name="z_phylo")
    u_phylo = jnp.matmul(A_cholesky, z_phylo) * sd_phylo
    z_obs = m.dist.normal(jnp.zeros(200), 1.0, name="z_obs")
    u_obs = z_obs * sd_obs
    mu = intercept + u_phylo[phylo_idx] + u_obs[obs_idx]
    m.dist.normal(mu, se, name="Y", obs=Zr)

m.fit(model, num_samples=1000, num_warmup=1000, num_chains=1, seed=0, progress_bar=False)
```
