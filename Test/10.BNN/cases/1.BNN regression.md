# 1. BNN regression (two layers, tanh hidden units, BNN.json)

Same model as `Documentation/33. BNN.qmd` and `../code/BNN regressions.py` (non-linear data), on `data/BNN.json`. Only `sigma` is compared (`params:`): the weights `w1`, `w2` are identified only up to permutation and sign of the hidden units.

```python
from BayesForge import bf
import json
import jax.numpy as jnp

m = bf(platform='cpu')

with open('BNN.json', 'r', encoding='utf-8') as file:
    # Load the JSON data into a Python dictionary
    data = json.load(file)
# X is already scaled
X = jnp.array(data['X']) # Note X shape = (N,2) where first column is the intercept and second column is the predictor
Y = jnp.array(data['Y']) # Note Y shape = (N,1) where N is the number of observations

m.data_on_model = dict(X = X, Y = Y)
# Define model ------------------------------------------------
def model(X, Y,  D_H=5, D_Y=1):  
    N, D_X = X.shape
    
    # First hidden layer: Transforms input to N × D_H (hidden units)
    w1 = m.bnn.layer_linear(
        X, 
        dist=m.dist.normal(
            0, 1,  name='w1',shape=(D_X,D_H)
            ),
        activation='tanh'
        )

    # sample final layer of weights and neural network output
    # Final layer (z3) computes linear combination of second hidden layer
    w2 = m.bnn.layer_linear(
        X=w1, 
        dist=m.dist.normal(0, 1,  name='w2',shape=(D_H,D_Y))
        )

    sigma = m.dist.exponential(1, name='sigma')

    m.dist.normal(w2, sigma, obs=Y,name='Y')

m.fit(model, num_samples=1000, num_warmup=1000, num_chains=4, seed=0, progress_bar=False)
```

library(BayesForge)
m <- importBF(platform = "cpu")
jnp <- reticulate::import("jax.numpy")

# Import data ------------------------------------------------
data <- jsonlite::fromJSON("BNN.json")
X <- jnp$array(data$X) # Note X shape = (N,2) where first column is the intercept and second column is the predictor

m$data_on_model <- list(X = X, Y = Y)

# Define model ------------------------------------------------
model <- function(X, Y, D_H = 5L, D_Y = 1L) {
  D_X <- X$shape[[2]]

  # First hidden layer: Transforms input to N × D_H (hidden units)
  w1 <- m$bnn$layer_linear(
    X,
    dist = m$dist$normal(0, 1, name = "w1", shape = reticulate::tuple(D_X, D_H)),
    activation = "tanh"
  )

  # Final layer: linear combination of the hidden units
  w2 <- m$bnn$layer_linear(
    X = w1,
    dist = m$dist$normal(0, 1, name = "w2", shape = reticulate::tuple(D_H, D_Y))
  )

  sigma <- m$dist$exponential(1, name = "sigma")

  m$dist$normal(w2, sigma, obs = Y, name = "Y")
}

m$fit(model, num_samples = 1000L, num_warmup = 1000L, num_chains = 4L, seed = 0L, progress_bar = FALSE)
