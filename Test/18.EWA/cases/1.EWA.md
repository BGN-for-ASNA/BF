# 1. EWA social learning (Panama capuchins)

Same model as section "1. Social Learning (EWA)" of `Documentation/32. EWA.qmd` (see also `../BF_backend/model.py`); only model-level parameters are compared.

```python
from BayesForge import bf
import jax.numpy as jnp
import jax

m = bf(platform='cpu')

# Data: capuchin foraging choices (panama_ewa) ------------------------------
import pandas as pd
d = pd.read_csv(m.load.panama_ewa(only_path=True))
K = 7  # foraging techniques

def cue(prefix, scale=False):
    """N x K matrix of the columns prefix1..prefix7, optionally divided by its maximum."""
    x = d[[f"{prefix}{k}" for k in range(1, K + 1)]].values
    return jnp.array(x / x.max() if scale else x)

y = cue("y")                                   # personal yield of each technique
y_prev = jnp.zeros_like(y).at[1:].set(y[:-1])  # yields of the previous row

m.data_on_model = dict(
    K=K,
    J=int(d["mono_index"].nunique()),            # number of monkeys
    tech=jnp.array(d["tech_index"].values - 1),  # chosen technique, 0-indexed
    id=jnp.array(d["mono_index"].values - 1),    # monkey, 0-indexed
    bout=jnp.array(d["forg_bout"].values),       # 1 = first bout (attractions reset)
    y_prev=y_prev,
    s=cue("s"),                                  # frequency cue
    ps=cue("ps", scale=True),                    # payoff cue
)

def model(K, J, tech, id, bout, y_prev, s, ps):
    # Priors
    lam   = m.dist.exponential(name="lambda", rate=1.0)
    mu    = m.dist.normal(name="mu", loc=jnp.zeros(4), scale=1.0)
    sigma = m.dist.exponential(name="sigma", rate=jnp.full(4, 3.0))
    L_Rho = m.dist.lkj_cholesky(name="L_Rho", dimension=4, concentration=3.0)
    z     = m.dist.normal(name="z", loc=jnp.zeros((4, J)), scale=1.0)

    # Varying effects
    L_scaled = L_Rho * sigma[:, None]
    a_id     = (L_scaled @ z).T

    # Parameters per individual
    phi_arr   = jax.nn.sigmoid(mu[0] + a_id[id, 0])      # Learning rate
    gamma_arr = jax.nn.sigmoid(mu[1] + a_id[id, 1])      # Social weight
    fconf_arr = jnp.exp(mu[2] + a_id[id, 2])             # Conformity
    Bpay_arr  = mu[3] + a_id[id, 3]                      # Pay-off bias

    # Scan carry: Attractions
    def step_fn(carry, x):
        AC = carry
        (phi_i, gamma_i, fconf_i, Bpay_i, tech_i, id_i, bout_i, prev_y_i, s_i, ps_i) = x

        # Update attractions (EWA)
        ac_new = jnp.where(bout_i == 1, jnp.zeros(K), (1.0 - phi_i) * AC[id_i] + phi_i * prev_y_i)

        # Asocial choice probability
        prob_individual = jax.nn.softmax(lam * ac_new)

        # Social choice probability (Pay-off bias)
        lin_mod = jnp.concatenate([jnp.ones(1), jnp.exp(Bpay_i * ps_i[1:])]) * jnp.power(s_i, fconf_i)
        safe_denom = jnp.where(jnp.sum(lin_mod) > 0, jnp.sum(lin_mod), 1.0)
        prob_social = lin_mod / safe_denom

        # Mixture probability
        prob_mix = (1.0 - gamma_i) * prob_individual + gamma_i * prob_social
        has_social_info = jnp.logical_and(bout_i > 1, jnp.sum(s_i) > 0)
        prob_final = jnp.where(has_social_info, prob_mix, prob_individual)

        return AC.at[id_i].set(ac_new), prob_final

    AC_init = jnp.zeros((J, K))
    xs = (phi_arr, gamma_arr, fconf_arr, Bpay_arr, tech, id, bout, y_prev, s, ps)
    _, probs_matrix = jax.lax.scan(step_fn, AC_init, xs)

    m.dist.categorical(name="obs", probs=probs_matrix, obs=tech)

m.fit(model, num_samples=1000, num_warmup=1000, num_chains=4, seed=0, progress_bar=False)
```

```r
library(BayesForge)
m <- importBF(platform = "cpu")
jnp <- reticulate::import("jax.numpy")
jax <- reticulate::import("jax")
py <- reticulate::import_builtins()

# Data: capuchin foraging choices (panama_ewa) ------------------------------
d <- read.csv(m$load$panama_ewa(only_path = TRUE))
K <- 7L  # foraging techniques

# N x K matrix of the columns prefix1..prefix7, optionally divided by its maximum
cue <- function(prefix, scale = FALSE) {
  x <- as.matrix(d[, paste0(prefix, 1:K)])
  jnp$array(if (scale) x / max(x) else x)
}

y <- as.matrix(d[, paste0("y", 1:K)])
y_prev <- rbind(0, y[-nrow(y), ])  # yields of the previous row

m$data_on_model <- list(
  K = K,
  J = length(unique(d$mono_index)),               # number of monkeys
  tech = jnp$array(as.integer(d$tech_index - 1)),  # chosen technique, 0-indexed
  id = jnp$array(as.integer(d$mono_index - 1)),    # monkey, 0-indexed
  bout = jnp$array(as.integer(d$forg_bout)),       # 1 = first bout (attractions reset)
  y_prev = jnp$array(y_prev),
  s = cue("s"),                                    # frequency cue
  ps = cue("ps", scale = TRUE)                     # payoff cue
)

# x[1:] of a JAX vector (all techniques but the first)
rest <- function(x) x$`__getitem__`(py$slice(1L, NULL))

model <- function(K, J, tech, id, bout, y_prev, s, ps) {
  # Priors
  lam   <- m$dist$exponential(name = "lambda", rate = 1.0)
  mu    <- m$dist$normal(name = "mu", loc = jnp$zeros(4L), scale = 1.0)
  sigma <- m$dist$exponential(name = "sigma", rate = jnp$full(4L, 3.0))
  L_Rho <- m$dist$lkj_cholesky(name = "L_Rho", dimension = 4L, concentration = 3.0)
  z     <- m$dist$normal(name = "z", loc = jnp$zeros(reticulate::tuple(4L, J)), scale = 1.0)

  # Varying effects
  L_scaled <- L_Rho * jnp$expand_dims(sigma, -1L)
  a_id     <- jnp$transpose(jnp$matmul(L_scaled, z))

  # Parameters per individual
  phi_arr   <- jax$nn$sigmoid(mu[0L] + a_id[id, 0L])  # Learning rate
  gamma_arr <- jax$nn$sigmoid(mu[1L] + a_id[id, 1L])  # Social weight
  fconf_arr <- jnp$exp(mu[2L] + a_id[id, 2L])         # Conformity
  Bpay_arr  <- mu[3L] + a_id[id, 3L]                  # Pay-off bias

  # Scan carry: Attractions
  step_fn <- function(carry, x) {
    AC <- carry
    phi_i <- x[[1]]; gamma_i <- x[[2]]; fconf_i <- x[[3]]; Bpay_i <- x[[4]]
    tech_i <- x[[5]]; id_i <- x[[6]]; bout_i <- x[[7]]; prev_y_i <- x[[8]]
    s_i <- x[[9]]; ps_i <- x[[10]]

    # Update attractions (EWA)
    ac_new <- jnp$where(bout_i == 1L, jnp$zeros(K), (1.0 - phi_i) * AC[id_i] + phi_i * prev_y_i)

    # Asocial choice probability
    prob_individual <- jax$nn$softmax(lam * ac_new)

    # Social choice probability (Pay-off bias)
    lin_mod <- jnp$concatenate(list(jnp$ones(1L), jnp$exp(Bpay_i * rest(ps_i)))) * jnp$power(s_i, fconf_i)
    safe_denom <- jnp$where(jnp$sum(lin_mod) > 0, jnp$sum(lin_mod), 1.0)
    prob_social <- lin_mod / safe_denom

    # Mixture probability
    prob_mix <- (1.0 - gamma_i) * prob_individual + gamma_i * prob_social
    has_social_info <- jnp$logical_and(bout_i > 1L, jnp$sum(s_i) > 0)
    prob_final <- jnp$where(has_social_info, prob_mix, prob_individual)

    reticulate::tuple(AC$at[id_i]$set(ac_new), prob_final)
  }

  AC_init <- jnp$zeros(reticulate::tuple(J, K))
  xs <- reticulate::tuple(phi_arr, gamma_arr, fconf_arr, Bpay_arr, tech, id, bout, y_prev, s, ps)
  probs_matrix <- jax$lax$scan(step_fn, AC_init, xs)[[2]]

  m$dist$categorical(name = "obs", probs = probs_matrix, obs = tech)
}

m$fit(model, num_samples = 1000L, num_warmup = 1000L, num_chains = 4L, seed = 0L, progress_bar = FALSE)
```
